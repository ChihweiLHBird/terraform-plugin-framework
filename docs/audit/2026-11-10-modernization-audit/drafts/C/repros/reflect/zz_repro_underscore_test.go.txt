package reflect_test

import (
	"context"
	"fmt"
	"testing"

	"github.com/hashicorp/terraform-plugin-go/tftypes"

	"github.com/hashicorp/terraform-plugin-framework/attr"
	"github.com/hashicorp/terraform-plugin-framework/internal/fwschema"
	refl "github.com/hashicorp/terraform-plugin-framework/internal/reflect"
	"github.com/hashicorp/terraform-plugin-framework/path"
	"github.com/hashicorp/terraform-plugin-framework/types"
)

type underscoreModel struct {
	Internal types.String `tfsdk:"_internal"`
}

// C3: schema names may start with "_", tfsdk tags may not.
func TestReproUnderscoreTag(t *testing.T) {
	ctx := context.Background()
	fmt.Println("schema name validation diags for _internal:", fwschema.IsValidAttributeName("_internal", path.Root("_internal")))
	objType := types.ObjectType{AttrTypes: map[string]attr.Type{"_internal": types.StringType}}
	obj := tftypes.NewValue(objType.TerraformType(ctx), map[string]tftypes.Value{"_internal": tftypes.NewValue(tftypes.String, "v")})
	var m underscoreModel
	for _, d := range refl.Into(ctx, objType, obj, &m, refl.Options{}, path.Empty()) {
		fmt.Printf("Into: %s | %s\n", d.Summary(), d.Detail())
	}
	_, diags := refl.FromValue(ctx, objType, underscoreModel{Internal: types.StringValue("v")}, path.Empty())
	for _, d := range diags {
		fmt.Printf("FromValue: %s | %s\n", d.Summary(), d.Detail())
	}
}
