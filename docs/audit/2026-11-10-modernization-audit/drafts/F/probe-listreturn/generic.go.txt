// Minimal "fixed type" custom Set/Map/Object types (value Type() is fixed, not derived).
type FixedSetType struct{ basetypes.SetType }

func (t FixedSetType) Equal(o attr.Type) bool { _, ok := o.(FixedSetType); return ok }
func (t FixedSetType) String() string        { return "FixedSetType" }
func (t FixedSetType) ValueFromSet(ctx context.Context, in basetypes.SetValue) (basetypes.SetValuable, diag.Diagnostics) {
	return FixedSetValue{SetValue: in}, nil
}
func (t FixedSetType) ValueFromTerraform(ctx context.Context, in tftypes.Value) (attr.Value, error) {
	v, err := t.SetType.ValueFromTerraform(ctx, in)
	if err != nil {
		return nil, err
	}
	sv, _ := v.(basetypes.SetValue)
	return FixedSetValue{SetValue: sv}, nil
}
func (t FixedSetType) ValueType(ctx context.Context) attr.Value { return FixedSetValue{} }

type FixedSetValue struct{ basetypes.SetValue }

func (v FixedSetValue) Type(ctx context.Context) attr.Type {
	return FixedSetType{SetType: basetypes.SetType{ElemType: types.StringType}}
}
func (v FixedSetValue) Equal(o attr.Value) bool { other, ok := o.(FixedSetValue); return ok && v.SetValue.Equal(other.SetValue) }

// errProv: function whose Run only returns an argument error (no Set).
type errProv struct{ ret function.Return }

func (p errProv) Metadata(_ context.Context, _ provider.MetadataRequest, resp *provider.MetadataResponse) { resp.TypeName = "p" }
func (p errProv) Schema(context.Context, provider.SchemaRequest, *provider.SchemaResponse)          {}
func (p errProv) Configure(context.Context, provider.ConfigureRequest, *provider.ConfigureResponse)  {}
func (p errProv) DataSources(context.Context) []func() datasource.DataSource                       { return nil }
func (p errProv) Resources(context.Context) []func() resource.Resource                             { return nil }
func (p errProv) Functions(context.Context) []func() function.Function {
	return []func() function.Function{func() function.Function { return errFunc{ret: p.ret} }}
}

type errFunc struct{ ret function.Return }

func (f errFunc) Metadata(_ context.Context, _ function.MetadataRequest, resp *function.MetadataResponse) { resp.Name = "f" }
func (f errFunc) Definition(_ context.Context, _ function.DefinitionRequest, resp *function.DefinitionResponse) {
	resp.Definition = function.Definition{Return: f.ret}
}
func (f errFunc) Run(ctx context.Context, req function.RunRequest, resp *function.RunResponse) {
	resp.Error = function.NewFuncError("boom")
}

func errE2E(name string, ret function.Return) {
	defer func() {
		if r := recover(); r != nil {
			fmt.Printf("%-48s PANIC: %v\n", name, r)
		}
	}()
	ctx := context.Background()
	srv5, _ := providerserver.NewProtocol5WithError(errProv{ret: ret})()
	srv6, _ := providerserver.NewProtocol6WithError(errProv{ret: ret})()
	func() {
		defer func() {
			if r := recover(); r != nil {
				fmt.Printf("%-48s v5 PANIC: %v\n", name, r)
			}
		}()
		r5, _ := srv5.CallFunction(ctx, &tfprotov5.CallFunctionRequest{Name: "f"})
		fmt.Printf("%-48s v5 err=%q\n", name, r5.Error.Text)
	}()
	r6, _ := srv6.CallFunction(ctx, &tfprotov6.CallFunctionRequest{Name: "f"})
	fmt.Printf("%-48s v6 err=%q\n", name, r6.Error.Text)
}

func runGeneric() {
	errE2E("ListReturn{ElementType}", function.ListReturn{ElementType: types.StringType})
	errE2E("ListReturn{CustomType}", function.ListReturn{CustomType: FixedListType{ListType: basetypes.ListType{ElemType: types.StringType}}})
	errE2E("SetReturn{CustomType}", function.SetReturn{CustomType: FixedSetType{SetType: basetypes.SetType{ElemType: types.StringType}}})
	errE2E("MapReturn{CustomType}", function.MapReturn{CustomType: FixedMapType{MapType: basetypes.MapType{ElemType: types.StringType}}})
	errE2E("ObjectReturn{CustomType}", function.ObjectReturn{CustomType: FixedObjectType{ObjectType: basetypes.ObjectType{AttrTypes: objAttrTypes}}})
	errE2E("StringReturn{}", function.StringReturn{})
}
