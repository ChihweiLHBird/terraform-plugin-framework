package main

import (
	"context"
	"fmt"

	"github.com/hashicorp/terraform-plugin-framework/attr"
	"github.com/hashicorp/terraform-plugin-framework/function"
	"github.com/hashicorp/terraform-plugin-framework/types"
	"github.com/hashicorp/terraform-plugin-framework/types/basetypes"
)

// PlainObjectType mirrors testtypes-style embedding without overrides.
type PlainObjectType struct{ basetypes.ObjectType }

func (t PlainObjectType) Equal(o attr.Type) bool {
	other, ok := o.(PlainObjectType)
	return ok && t.ObjectType.Equal(other.ObjectType)
}

type objModel struct {
	A types.String `tfsdk:"a"`
}

func runObject() {
	ctx := context.Background()
	ct := PlainObjectType{ObjectType: basetypes.ObjectType{AttrTypes: map[string]attr.Type{"a": types.StringType}}}
	r := function.ObjectReturn{CustomType: ct}
	rd, ferr := r.NewResultData(ctx)
	fmt.Printf("object plain: NewResultData err=%v value=%#v\n", ferr, rd.Value())
	fmt.Printf("object plain: Set(struct) err=%v\n", rd.Set(ctx, objModel{A: types.StringValue("x")}))
	safe("object plain e2e set", func() { e2e("object plain e2e set struct", r, objModel{A: types.StringValue("x")}) })
	safe("object plain e2e Run error", func() { errE2E("object plain e2e Run error", r) })
}
