func goldenDumpResp(b *strings.Builder, label string, resp *ModifyAttributePlanResponse) {
	fmt.Fprintf(b, "%s: plan=%T %v\n", label, resp.AttributePlan, resp.AttributePlan)
	fmt.Fprintf(b, "%s: diags(nil=%v len=%d)\n", label, resp.Diagnostics == nil, len(resp.Diagnostics))
	for _, d := range resp.Diagnostics {
		p := ""
		if dp, ok := d.(diag.DiagnosticWithPath); ok {
			p = dp.Path().String()
		}
		fmt.Fprintf(b, "%s:   %T %v %q %q path=%q\n", label, d, d.Severity(), d.Summary(), d.Detail(), p)
	}
	fmt.Fprintf(b, "%s: requiresReplace(nil=%v)=%v private(nil=%v)\n", label, resp.RequiresReplace == nil, resp.RequiresReplace, resp.Private == nil)
}

func goldenLogs(b *strings.Builder, label string, out *bytes.Buffer) {
	entries, err := tfsdklogtest.MultilineJSONDecode(out)
	if err != nil {
		panic(err)
	}
	for _, e := range entries {
		keys := make([]string, 0, len(e))
		for k := range e {
			if k == "@caller" || k == "@timestamp" {
				continue
			}
			keys = append(keys, k)
		}
		sort.Strings(keys)
		fmt.Fprintf(b, "%s: log", label)
		for _, k := range keys {
			fmt.Fprintf(b, " %s=%v", k, e[k])
		}
		fmt.Fprintln(b)
	}
}
