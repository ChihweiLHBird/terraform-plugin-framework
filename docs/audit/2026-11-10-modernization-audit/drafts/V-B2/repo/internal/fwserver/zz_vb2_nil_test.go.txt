package fwserver_test

import (
	"context"
	"fmt"
	"testing"

	"github.com/hashicorp/terraform-plugin-go/tftypes"

	dsschema "github.com/hashicorp/terraform-plugin-framework/datasource/schema"
	"github.com/hashicorp/terraform-plugin-framework/internal/fwserver"
	"github.com/hashicorp/terraform-plugin-framework/list"
	"github.com/hashicorp/terraform-plugin-framework/provider"
	"github.com/hashicorp/terraform-plugin-framework/resource"
	"github.com/hashicorp/terraform-plugin-framework/resource/identityschema"
	"github.com/hashicorp/terraform-plugin-framework/resource/schema"
)

type vbDeferProv struct{ vbAliasProvider }

func (vbDeferProv) Configure(_ context.Context, _ provider.ConfigureRequest, resp *provider.ConfigureResponse) {
	resp.Deferred = &provider.Deferred{Reason: provider.DeferredReasonProviderConfigUnknown}
}

type vbCreateRes struct{}

func (vbCreateRes) Metadata(context.Context, resource.MetadataRequest, *resource.MetadataResponse) {}
func (vbCreateRes) Schema(context.Context, resource.SchemaRequest, *resource.SchemaResponse)       {}
func (vbCreateRes) Create(_ context.Context, _ resource.CreateRequest, resp *resource.CreateResponse) {
	resp.State.Raw = tftypes.NewValue(tftypes.Object{AttributeTypes: map[string]tftypes.Type{"id": tftypes.String}}, map[string]tftypes.Value{"id": tftypes.NewValue(tftypes.String, "x")})
}
func (vbCreateRes) Read(context.Context, resource.ReadRequest, *resource.ReadResponse)       {}
func (vbCreateRes) Update(context.Context, resource.UpdateRequest, *resource.UpdateResponse) {}
func (vbCreateRes) Delete(context.Context, resource.DeleteRequest, *resource.DeleteResponse) {}

type vbList struct{}

func (vbList) Metadata(context.Context, resource.MetadataRequest, *resource.MetadataResponse) {}
func (vbList) ListResourceConfigSchema(context.Context, list.ListResourceSchemaRequest, *list.ListResourceSchemaResponse) {
}
func (vbList) List(_ context.Context, req list.ListRequest, _ *list.ListResultsStream) {
	fmt.Printf("VB2 list provider saw Config.Schema type %T\n", req.Config.Schema)
}

func vbCatch(t *testing.T, name string, f func()) {
	defer func() {
		if r := recover(); r != nil {
			t.Logf("B2-6 %s: PANIC %v", name, r)
		}
	}()
	f()
	t.Logf("B2-6 %s: no panic", name)
}

func TestVB2Nil(t *testing.T) {
	ctx := context.Background()
	vbCatch(t, "(1) ConfigureProvider(nil req, deferring provider)", func() {
		s := &fwserver.Server{Provider: vbDeferProv{}}
		s.ConfigureProvider(ctx, nil, &provider.ConfigureResponse{})
	})
	vbCatch(t, "(2) ReadDataSource(nil Config)", func() {
		s := &fwserver.Server{Provider: vbAliasProvider{}}
		s.ReadDataSource(ctx, &fwserver.ReadDataSourceRequest{DataSource: vbDS{"x"}, DataSourceSchema: dsschema.Schema{}}, &fwserver.ReadDataSourceResponse{})
	})
	sch := schema.Schema{Attributes: map[string]schema.Attribute{"id": schema.StringAttribute{Computed: true}}}
	vbCatch(t, "(3) ApplyResourceChange(nil PriorState, nil PlannedState)", func() {
		s := &fwserver.Server{Provider: vbAliasProvider{}}
		s.ApplyResourceChange(ctx, &fwserver.ApplyResourceChangeRequest{ResourceSchema: sch, Resource: vbCreateRes{}}, &fwserver.ApplyResourceChangeResponse{})
	})
	ids := identityschema.Schema{Attributes: map[string]identityschema.Attribute{"id": identityschema.StringAttribute{RequiredForImport: true}}}
	vbCatch(t, "(4) ListResource(nil Config, nil ResourceSchema, identity schema)", func() {
		s := &fwserver.Server{Provider: vbAliasProvider{}}
		s.ListResource(ctx, &fwserver.ListRequest{ListResource: vbList{}, ResourceIdentitySchema: ids}, &fwserver.ListResultsStream{})
	})
	vbCatch(t, "(4b) ListResource(nil Config, ResourceSchema set)", func() {
		s := &fwserver.Server{Provider: vbAliasProvider{}}
		s.ListResource(ctx, &fwserver.ListRequest{ListResource: vbList{}, ResourceSchema: sch, ResourceIdentitySchema: ids}, &fwserver.ListResultsStream{})
	})
}
